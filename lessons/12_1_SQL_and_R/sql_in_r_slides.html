# Create an in-memory SQLite database
con <- dbConnect(RSQLite::SQLite(), ":memory:")

# Write our flights data to the database

# Let's also add related tables for richer queries

  SELECT a.name as airline_name, COUNT(*) as flights

🆚 dplyr vs SQL: Same Results, Different Syntax

  • SQL Approach
  • dplyr Approach
# SQL way
sql_result <- dbGetQuery(con, "
  SELECT dest, COUNT(*) as n_flights, AVG(distance) as avg_distance
  FROM flights 
  WHERE distance > 1000
  GROUP BY dest
  HAVING n_flights > 100
  ORDER BY avg_distance DESC
  LIMIT 5")
# dplyr way
dplyr_result <- flights %>%
  filter(distance > 1000) %>%
  group_by(dest) %>%
  summarise(
    n_flights = n(),
    avg_distance = mean(distance, na.rm = TRUE),
    .groups = "drop"
  ) %>%
  filter(n_flights > 100) %>%
  arrange(desc(avg_distance)) %>%
  slice_head(n = 5)

Important

Both produce identical results! Choose what feels more natural 🎯

  • 📊 Scale matters: Performance gap widens with data size
